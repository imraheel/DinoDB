# NexDB Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build NexDB — a cross-platform, RoboMongo-style MongoDB client (Electron + React + pnpm) with multiple SQLite-stored connections, multi-tab collection browsing, and one Electron `UtilityProcess` per tab so a stuck tab never blocks other tabs or the main process.

**Architecture:** Single renderer window (React + Context/useReducer) never touches MongoDB/SQLite — it calls a preload `window.api`. The main process owns SQLite repos and relays typed requests to per-connection "session" utility processes and per-tab "tab" utility processes (correlation-id + hard timeout, so every request settles). The `mongodb` driver runs only inside utility processes; documents cross process boundaries as relaxed Extended JSON (plain, structured-clone-safe objects).

**Tech Stack:** Electron (`utilityProcess.fork`), electron-vite, React 19, Tailwind CSS v4, better-sqlite3, mongodb driver v6, Monaco editor, vitest (run under `ELECTRON_RUN_AS_NODE` for native-ABI tests), electron-builder, pnpm.

Spec: `docs/superpowers/specs/2026-10-09-nexdb-design.md` (read it first).

### `src/shared/mongo-uri.ts`

```ts
import type { ConnectionConfig } from './types';

export type UriSource = Pick<ConnectionConfig,
  'host' | 'port' | 'defaultDatabase' | 'username' | 'password' |
  'authMechanism' | 'authSource' | 'authMechanismProperties' | 'extraOptions'>;

export function isJsonObject(text: string): boolean {
  try {
    const value: unknown = JSON.parse(text);
    return typeof value === 'object' && value !== null && !Array.isArray(value);
  } catch { return false; }
}

export function buildMongoUri(cfg: UriSource): string {
  const auth = cfg.username
    ? `${encodeURIComponent(cfg.username)}${cfg.password ? `:${encodeURIComponent(cfg.password)}` : ''}@`
    : '';
  const params: string[] = [];

  if (cfg.username) {
    if (cfg.authSource) params.push(`authSource=${encodeURIComponent(cfg.authSource)}`);
    if (cfg.authMechanism && cfg.authMechanism !== 'DEFAULT') params.push(`authMechanism=${cfg.authMechanism}`);
    const props = cfg.authMechanismProperties.trim();
    if (props) {
      if (!isJsonObject(props)) throw new Error('authMechanismProperties must be a JSON object');
      const pairs = Object.entries(JSON.parse(props) as Record<string, unknown>).map(([k, v]) => `${k}:${String(v)}`);
      params.push(`authMechanismProperties=${encodeURIComponent(pairs.join(','))}`);
    }
  }

  const extra = cfg.extraOptions.trim();
  if (extra) {
    if (!isJsonObject(extra)) throw new Error('extraOptions must be a JSON object');
    for (const [k, v] of Object.entries(JSON.parse(extra) as Record<string, unknown>)) {
      if (!/^[A-Za-z][A-Za-z0-9]*$/.test(k)) throw new Error(`Invalid extra option name: ${k}`);
      params.push(`${k}=${encodeURIComponent(String(v))}`);
    }
  }

  const path = cfg.defaultDatabase ? `/${cfg.defaultDatabase}` : '/';
  const query = params.length ? `?${params.join('&')}` : '';
  return `mongodb://${auth}${cfg.host}:${cfg.port}${path}${query}`;
}
```

### `src/shared/validate.ts`

```ts
import { buildMongoUri, isJsonObject } from './mongo-uri';
import type { ConnectionInput } from './types';

export function validateConnection(input: ConnectionInput): string[] {
  const errors: string[] = [];
  if (!input.name.trim()) errors.push('Name is required');
  if (!input.host.trim()) errors.push('Host is required');
  if (!Number.isInteger(input.port) || input.port < 1 || input.port > 65535) {
    errors.push('Port must be an integer between 1 and 65535');
  }
  if (input.authMechanismProperties.trim() && !isJsonObject(input.authMechanismProperties.trim())) {
    errors.push('authMechanismProperties must be a JSON object');
  }
  if (input.extraOptions.trim() && !isJsonObject(input.extraOptions.trim())) {
    errors.push('extraOptions must be a JSON object');
  }
  if (errors.length === 0) {
    try { buildMongoUri(input); } catch (err) { errors.push(err instanceof Error ? err.message : String(err)); }
  }
  return errors;
}
```

### `src/main/services/utility-client.ts` (Task 4 — authoritative)

```ts
import type { InitMessage, MainToUtility, UtilityRequest, UtilityResponses, UtilityToMain } from '../../shared/ipc';

export interface ForkedProcess {
  postMessage(message: MainToUtility): void;
  on(event: 'message', listener: (message: UtilityToMain) => void): void;
  on(event: 'exit', listener: (code: number | null) => void): void;
  kill(): boolean;
}

export interface UtilityClientDeps {
  resolveScriptPath: () => string;
  fork: (scriptPath: string) => ForkedProcess;
}

interface PendingRequest {
  resolve: (value: unknown) => void;
  reject: (error: Error) => void;
  timer: ReturnType<typeof setTimeout>;
}

export interface RequestOptions { timeoutMs: number; maxTimeMs: number }

export class UtilityClient {
  private process: ForkedProcess | null = null;
  private nextId = 1;
  private readonly pending = new Map<number, PendingRequest>();

  constructor(private readonly label: string, private readonly deps: UtilityClientDeps) {}

  async start(init: InitMessage, timeoutMs = 20000): Promise<void> {
    if (this.process) throw new Error(`Utility client "${this.label}" already started`);
    const proc = this.deps.fork(this.deps.resolveScriptPath());
    this.process = proc;

    return new Promise<void>((resolve, reject) => {
      let settled = false;
      const timer = setTimeout(() => {
        if (settled) return;
        settled = true;
        proc.kill();
        this.process = null;
        reject(new Error(`Connection timed out after ${timeoutMs}ms`));
      }, timeoutMs);

      const settleStart = (error?: Error) => {
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        if (error) reject(error); else resolve();
      };

      proc.on('message', (message) => {
        if (message.kind === 'ready') settleStart();
        else if (message.kind === 'fatal') { settleStart(new Error(message.error)); this.failAll(new Error(message.error)); }
        else if (message.kind === 'response') {
          const request = this.pending.get(message.id);
          if (!request) return;
          this.pending.delete(message.id);
          clearTimeout(request.timer);
          if (message.ok) request.resolve(message.result);
          else request.reject(new Error(message.error));
        }
      });

      proc.on('exit', (code) => {
        const error = new Error(`Utility process "${this.label}" exited (code ${code ?? 'unknown'})`);
        settleStart(error);
        this.failAll(error);
        this.process = null;
      });

      proc.postMessage({ kind: 'init', init });
    });
  }

  async request<K extends UtilityRequest['kind']>(
    request: Extract<UtilityRequest, { kind: K }>,
    options: RequestOptions
  ): Promise<UtilityResponses[K]> {
    if (!this.process) throw new Error(`Utility client "${this.label}" is not running`);
    const id = this.nextId++;
    const proc = this.process;

    return new Promise<UtilityResponses[K]>((resolve, reject) => {
      const timer = setTimeout(() => {
        this.pending.delete(id);
        reject(new Error(`Operation timed out after ${options.timeoutMs}ms`));
      }, options.timeoutMs);

      this.pending.set(id, { resolve: (value) => resolve(value as UtilityResponses[K]), reject, timer });
      proc.postMessage({ kind: 'request', id, request, maxTimeMs: options.maxTimeMs });
    });
  }

  private failAll(error: Error): void {
    for (const [, entry] of this.pending) { clearTimeout(entry.timer); entry.reject(error); }
    this.pending.clear();
  }

  dispose(): void {
    this.process?.kill();
    this.process = null;
    this.failAll(new Error(`Utility client "${this.label}" disposed`));
  }
}
```
